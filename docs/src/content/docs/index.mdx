import { Card, CardGrid } from "@astrojs/starlight/components";

<CardGrid stagger>
  <Card title="The setup is on us" icon="setting">
    Command buses, aggregate repositories, event dispatchers and projection runners are wired for you. You write handlers, appliers and projections.
  </Card>
  <Card title="Types inferred, never written" icon="approve-check">
    Payloads, state and query results are typed from your code. Change a field and the compiler points at every place that breaks.
  </Card>
  <Card title="Processes are first-class" icon="random">
    A process follows one aggregate instance over time: it says which events open and close it, keeps state between them, and runs `on-timeout.ts` when its deadline passes. Retries and dead letters are the runtime's job.
  </Card>
  <Card title="One database is all your infra" icon="seti:db">
    Events, read models, scheduling and dead letters on a single Postgres or SQLite. No brokers, no queues to operate. When a projection goes wrong, `bounda rebuild`; when a policy dies, `bounda dead-letters`; when a payload changes shape, an upcast next to the event.
  </Card>
</CardGrid>

<Aside type="caution" title="Alpha">
  Every published version is a prerelease, so a plain install gets one: `npm create bounda@latest`,
  `npm install @bounda-dev/core`. The API can change between alphas without a deprecation cycle,
  and [Deployment](/guides/deployment/) lists what a production app does not have yet.
</Aside>
